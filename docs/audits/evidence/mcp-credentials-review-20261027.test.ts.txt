import assert from "node:assert/strict";
import { mock, test } from "node:test";
import { NextRequest } from "next/server";

const actor = "isolated-audit-user";
type TokenRow = { userId: string; tokenHash: string; name: string; createdAt: Date; revokedAt: Date | null; lastUsedAt: Date | null };
let row: TokenRow | null = null;
const originalDate = new Date("2026-09-30T18:15:08.658Z");
const prisma = { mcpApiToken: {
  findUnique: async ({ where }: { where: { userId: string } }) => row?.userId === where.userId ? { ...row } : null,
  upsert: async ({ create, update }: { create: Partial<TokenRow>; update: Partial<TokenRow> }) => {
    row = row ? { ...row, ...update } : { ...create, createdAt: originalDate, revokedAt: null, lastUsedAt: null } as TokenRow;
    return row;
  },
  findFirst: async ({ where }: { where: { tokenHash: string; revokedAt: null } }) => row?.tokenHash === where.tokenHash && row.revokedAt === null ? { userId: row.userId } : null,
  update: async ({ data }: { data: Partial<TokenRow> }) => { if (row) Object.assign(row, data); return row; },
  updateMany: async ({ data }: { data: Partial<TokenRow> }) => { if (row) Object.assign(row, data); return { count: row ? 1 : 0 }; },
} };
mock.module("@auto-articulos/db", { namedExports: { prisma } });
mock.module("../current-user.ts", { namedExports: { getCurrentUserId: async () => actor } });

test("isolated audit of actual credential routes and middleware; no external I/O", async (t) => {
  const issuer = await import("../../app/api/configuracion/mcp-token/route");
  const lookup = await import("../../app/api/mcp/token-lookup/route");
  const { middleware } = await import("../../middleware");
  let credential = "";
  const issue = () => issuer.POST(new Request("https://redes.lasolucionweb.net/api/configuracion/mcp-token", { method: "POST", body: JSON.stringify({ name: "Isolated audit" }) }));
  const resolve = (token: string) => lookup.POST(new NextRequest("https://redes.lasolucionweb.net/api/mcp/token-lookup", { method: "POST", headers: { authorization: `Bearer ${token}` } }));
  let internalStatus: number | null = null;
  const originalFetch = globalThis.fetch;
  globalThis.fetch = async (input, init) => {
    const url = new URL(String(input));
    assert.equal(url.pathname, "/api/mcp/token-lookup", "only internal lookup may be called");
    if (internalStatus) return new Response("service unavailable", { status: internalStatus });
    return lookup.POST(new NextRequest(url, init));
  };
  try {
    await t.test("issued token authenticates to the correct native user", async () => {
      const response = await issue();
      assert.equal(response.status, 200);
      credential = (await response.json()).token;
      assert.match(credential, /^sta_[A-Za-z0-9_-]{43}$/);
      const authenticated = await resolve(credential);
      assert.equal(authenticated.status, 200);
      assert.deepEqual(await authenticated.json(), { userId: actor });
    });
    await t.test("valid token survives middleware on both hosts and overwrites forged identity", async () => {
      for (const host of ["articulos.lasolucionweb.net", "redes.lasolucionweb.net"]) {
        const request = new NextRequest(`https://${host}/api/mcp`, { method: "POST", headers: { authorization: `Bearer ${credential}`, origin: "https://www.meta.ai", "x-user-id": "another-tenant" } });
        const response = await middleware(request);
        assert.equal(response.status, 200);
        assert.equal(response.headers.get("x-middleware-request-x-user-id"), actor);
      }
    });
    await t.test("unknown credential is denied instead of accepting a claimed identity", async () => {
      const response = await resolve("sta_deliberately-invalid");
      assert.equal(response.status, 401);
    });
    await t.test("CURRENT GAP: Redes MCP does not provide the trusted Hub product context", async () => {
      const response = await middleware(new NextRequest("https://redes.lasolucionweb.net/api/mcp", { method: "POST", headers: { authorization: `Bearer ${credential}` } }));
      assert.equal(response.status, 200);
      assert.equal(response.headers.get("x-middleware-request-x-hub-product-slug"), null);
    });
    await t.test("CURRENT BUG: generating again silently invalidates the previous credential", async () => {
      const previous = credential;
      const second = await issue();
      assert.equal(second.status, 200, "no replacement intent/confirmation was provided");
      credential = (await second.json()).token;
      assert.notEqual(credential, previous);
      assert.equal((await resolve(previous)).status, 401);
      assert.equal((await resolve(credential)).status, 200);
    });
    await t.test("CURRENT GAP: metadata preserves the first creation date after replacement", async () => {
      const metadata = await issuer.GET();
      const body = await metadata.json();
      assert.equal(body.createdAt, originalDate.toISOString());
      assert.equal(body.active, true);
      assert.equal("token" in body, false);
      assert.equal("tokenHash" in body, false);
    });
    await t.test("CURRENT GAP: lookup service outage is mislabeled as authentication 401", async () => {
      internalStatus = 503;
      const response = await middleware(new NextRequest("https://redes.lasolucionweb.net/api/mcp", { method: "POST", headers: { authorization: `Bearer ${credential}` } }));
      assert.equal(response.status, 401);
      internalStatus = null;
    });
    await t.test("explicit revocation denies the current credential", async () => {
      await issuer.DELETE();
      assert.equal((await resolve(credential)).status, 401);
    });
  } finally {
    globalThis.fetch = originalFetch;
    row = null;
    mock.restoreAll();
  }
});
